48def _brl(value: float | Decimal, decimals: int = 0) -> str: 

49 text = f"{float(value):,.{decimals}f}".replace(",", "X").replace(".", ",").replace("X", ".")

53def _pct(value: float | None, decimals: int = 1) -> str:

57def _style(ax: Any) -> None: 

58 for side in ("top", "right", "left"): 

59 ax.spines[side].set_visible(False) 

60 ax.spines["bottom"].set_color(GRID) 

61 ax.tick_params(colors=MUTED, labelsize=9, length=0)

66def _svg(fig: Any) -> str: 

67 buffer = io.StringIO() 

68 fig.savefig(buffer, format="svg", bbox_inches="tight", transparent=True) 

69 plt.close(fig) 

70 svg = buffer.getvalue() 

71 return svg[svg.index("<svg") :]

74def _mrr_chart(rows: list[tuple[Any, ...]]) -> str: 

75 if not rows: 75 ↛ 76line 75 didn't jump to line 76 because the condition on line 75 was never true

76 return '<p class="note">Ainda não há meses completos depois da base inicial.</p>' 

77 months = sorted({r[0] for r in rows}) 

78 fig, ax = plt.subplots(figsize=(10, 3.2)) 

79 width = 0.15 

80 kinds = [k for k in MOVEMENT_COLORS if any(r[1] == k for r in rows)] 

81 for i, kind in enumerate(kinds): 

82 values = [float(next((r[2] for r in rows if r[0] == m and r[1] == kind), 0)) for m in months] 

83 offsets = [x + (i - (len(kinds) - 1) / 2) * width for x in range(len(months))] 

84 ax.bar(offsets, values, width=width * 0.9, color=MOVEMENT_COLORS[kind], label=MOVEMENT_LABELS[kind]) 

85 ax.axhline(0, color=MUTED, linewidth=0.8) 

86 ax.set_xticks(range(len(months)), [f"{MONTHS_PT[m.month - 1]}/{m.year % 100}" for m in months]) 

87 ax.yaxis.set_major_formatter( 

88 FuncFormatter(lambda v, _: "0" if v == 0 else f"{v / 1000:,.0f} mil".replace(",", ".")) 

89 ) 

90 _style(ax) 

91 ax.legend(frameon=False, fontsize=8, ncol=len(kinds), loc="upper center", bbox_to_anchor=(0.5, 1.15)) 

92 return _svg(fig)

95def _hbar_chart(labels: list[str], values: list[float], color: str, fmt: str = "{:.0f}") -> str:

110def _table(headers: list[str], rows: list[list[str]]) -> str: 

111 head = "".join(f"<th>{escape(h)}</th>" for h in headers) 

112 body = "".join("<tr>" + "".join(f"<td>{escape(c)}</td>" for c in row) + "</tr>" for row in rows) 

113 return f"<table><thead><tr>{head}</tr></thead><tbody>{body}</tbody></table>"

145def build_report(database: Path, output: Path, *, revision: str | None = None) -> Path: 

146 con = duckdb.connect(str(database), read_only=True) 

147 try: 

148 kpi = con.execute("select * from marts.rpt_fleet_kpis").fetchone() 

149 cols = [d[0] for d in con.description or ()] 

150 k = dict(zip(cols, kpi or (), strict=False)) 

151 mrr_rows = con.execute( 

152 # O primeiro mês é a carteira inicial, não aquisição: fica fora do gráfico e vira a base. 

153 "select month_start, movement_type, sum(mrr_change) from marts.fct_mrr_movements " 

154 "where movement_type <> 'flat' and month_start > (select min(month_start) from marts.fct_mrr_movements) " 

155 "group by all order by 1" 

156 ).fetchall() 

157 base_month, base_mrr = con.execute( 

158 "select month_start, sum(mrr) from marts.fct_mrr_movements " 

159 "where month_start = (select min(month_start) from marts.fct_mrr_movements) group by 1" 

160 ).fetchone() or (None, 0) 

161 actions = con.execute( 

162 "select recommended_action, count(*) from marts.fct_asset_health " 

163 "where recommended_action <> 'ok' group by 1 order by 2 desc" 

164 ).fetchall() 

165 sla = con.execute( 

166 "select priority, avg(case when is_breached then 0 else 1 end), count(*) " 

167 "from marts.fct_ticket_sla group by 1 order by 1" 

168 ).fetchall() 

169 top_customers = con.execute( 

170 "select customer_name, segment, state, current_seats, current_mrr from marts.dim_customers " 

171 "where is_active order by current_mrr desc limit 8" 

172 ).fetchall() 

173 quality = con.execute( 

174 "select source, partitions, rows_read, rows_quarantined, quarantine_rate, orphan_rows " 

175 "from marts.rpt_data_quality order by source" 

176 ).fetchall() 

177 finally: 

178 con.close() 

179 

180 kpis = [ 

181 ("MRR", _brl(k["mrr"]), f"churn no último mês: {_brl(k['churned_mrr_last_month'])}"), 

182 ("Clientes ativos", str(k["active_customers"]), ""), 

183 ( 

184 "Equipamentos alocados", 

185 f"{k['assets_deployed']:,}".replace(",", "."), 

186 f"ocupação {_pct(k['fleet_utilization'])}", 

187 ), 

188 ("SLA de chamados (30d)", _pct(k["sla_compliance_30d"]), "resolvidos dentro do prazo"), 

189 ( 

190 "Devoluções", 

191 f"{k['return_median_cycle_days']:.1f} dias".replace(".", ","), 

192 f"{_pct(k['return_issue_rate'])} com avaria", 

193 ), 

194 ("Ações na frota", str(k["assets_needing_action"]), f"quarentena na ingestão: {_pct(k['quarantine_rate'], 2)}"), 

195 ] 

196 kpi_html = "".join( 

197 f'<div class="kpi"><span>{escape(t)}</span><strong>{escape(v)}</strong><small>{escape(s)}</small></div>' 

198 for t, v, s in kpis 

199 ) 

200 actions_svg = _hbar_chart([ACTION_LABELS.get(a, a) for a, _ in actions], [float(n) for _, n in actions], "#f59e0b") 

201 sla_svg = _hbar_chart( 

202 [f"{p} ({n} chamados)" for p, _, n in sla], [float(r) * 100 for _, r, _ in sla], "#2563eb", "{:.1f}%" 

203 ) 

204 customers_table = _table( 

205 ["Cliente", "Segmento", "UF", "Assentos", "MRR"], 

206 [[n, s, uf, str(seats), _brl(mrr)] for n, s, uf, seats, mrr in top_customers], 

207 ) 

208 quality_table = _table( 

209 ["Fonte", "Partições", "Linhas", "Quarentena", "Taxa", "Órfãos"], 

210 [[s, str(p), f"{r:,}".replace(",", "."), str(q), _pct(rate, 2), str(o)] for s, p, r, q, rate, o in quality], 

211 ) 

212 generated = datetime.now(UTC).strftime("%Y-%m-%d %H:%M UTC") 

213 base_label = f"{MONTHS_PT[base_month.month - 1]}/{base_month.year}" if base_month else "—" 

214 rev = f" · revisão <code>{escape(revision[:7])}</code>" if revision else "" 

215
